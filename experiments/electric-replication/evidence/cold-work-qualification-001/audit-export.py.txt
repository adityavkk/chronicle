def histogram(data):
    # ds-bench93a1a066 uses hdrhistogram7.5.4's uncompressed V2Serializer;
    # fixed numeric header plus canonical ZigZag/LEB128-64b9B count tokens.
    cookie,length,normal,digits,low,high,ratio=struct.unpack('>IIIIQQd',data[:40])
    assert (cookie,normal,digits,low,high,ratio)==(0x1c849313,0,3,1,60000000,1.0)
    assert len(data)==40+length and 1<=length<=9*17408
    offset,bins,total=40,0,0
    tokens=[]
    while offset<len(data):
        n=0;start=offset
        for i in range(9):
            b=data[offset];offset+=1
            n|=(b if i==8 else b&127)<<(7*i)
            if i==8 or b<128:break
        assert offset-start==max(1,min(9,(n.bit_length()+6)//7))
        value=(n>>1)^-(n&1)
        assert value != -(1<<63)
        if value<0:
            assert value<=-2
            bins-=value
        else:
            assert bins<17408
            bins+=1;total+=value
        assert bins<=17408 and total<(1<<64)
        tokens.append(value)
    assert offset==len(data)
    return json.dumps(dict(hdr_v2_numeric=True,digits=digits,low=low,high=high,total=total,tokens=tokens))

def raw(path):
    data=(ROOT/path).read_bytes()
    assert len(data)<50*1024**2,(path,'large stored file')
    if path.endswith('.gz'):return gzip.decompress(data)
    if path.endswith('.zst'):return subprocess.check_output(['zstd','-dq','-c',str(ROOT/path)])
    return data

def descriptor_keys(value,cluster):
    if isinstance(value,dict):
        if {'birth','len','sha256','kind'} <= value.keys():
            k=value['kind']
            if 'Inventory' in k:kind,inc,start='i',0,k['Inventory']['cut']
            else:
                item=k['Payload'];kind,inc,start='p',item['incarnation'],item['start']
            for group in (0,1):
                yield f"electric/{cluster}/{group}/{value['birth']:020}/{kind}/{inc:020}/{start:020}/{value['len']:020}/{value['sha256']}"
        else:
            for v in value.values():yield from descriptor_keys(v,cluster)
    elif isinstance(value,list):
        for v in value:yield from descriptor_keys(v,cluster)

def inspect(path,data,parent):
    global reviewed,events,decoded_bytes,histograms
    decoded_bytes+=len(data)
    assert not private.search(data),(path,'private endpoint/key/access-id pattern')
    assert not any(n in data for n in needles),(path,'retired fixture private key')
    assert not data.startswith((b'SQLite format 3\0',b'PERFILE2',b'\x7fELF')),(path,'raw store/profile/binary')
    digest=hashlib.sha256(data).hexdigest()
    if path.endswith('.hdr'):
        text=histogram(data);histograms+=1
    else:
        try:text=data.decode()
        except UnicodeDecodeError:raise AssertionError((path,'unknown binary format')) from None
    # No retained token fixture is expected in this increment. If one appears,
    # stop for its issuer/key-lifecycle/signature audit; do not assume JWT shape
    # or whitelist arbitrary strings merely because they came from a test.
    assert not re.search(r'Bearer\s+[A-Za-z0-9_-]{16,}\.[A-Za-z0-9_-]{16,}(?:\.[A-Za-z0-9_-]+)?',text,re.I),(path,'new bearer literal requires independent audit')
    if re.search(r'/objects-\d+\.jsonl(?:\.(?:gz|zst))?$',path):
        normalized=[]
        for line in text.splitlines():
            value=json.loads(line)
            inv=value.get('inventory')
            if inv:
                assert set(inv)=={'cut','records'}
                encoded=bytearray(b'ERIC0001'+inv['cut'].to_bytes(8,'big'))
                for r in inv['records']:
                    assert set(r)=={'birth','key_sha256'}
                    assert r['key_sha256'] in known[parent],(path,'unverified inventory key hash',r['birth'])
                    encoded.extend(r['birth'].to_bytes(8,'big')+bytes.fromhex(r['key_sha256']))
                    reviewed+=1
                assert hashlib.sha256(encoded).hexdigest()==value['sha256']==value['path'].rsplit('/',1)[-1]
                assert len(encoded)==value['bytes'] and value['method']=='GET'
                value['inventory']=dict(cut=inv['cut'],verified_public_key_hashes=len(inv['records']))
                events+=1
            normalized.append(json.dumps(value,separators=(',',':')))
        text='\n'.join(normalized)+'\n'
    output=OUT/'decoded'/(hashlib.sha256(text.encode()).hexdigest()+'.txt')
    output.write_text(text)
    rows.append(dict(path=path,decoded_sha256=digest,decoded_bytes=len(data),projection=str(output.relative_to(ROOT)),projection_sha256=output.stem))
