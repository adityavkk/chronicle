"""Fail-closed review of every finding in the decoded increment; print no secrets."""
from collections import defaultdict
import hashlib
import json
from pathlib import Path
import re

from cold_benchmark_check import read_object_events

ROOT=Path('/home/user/workspace/repo')
OUT=ROOT/'.tmp/cold-work-audit-003'
EVIDENCE=ROOT/'experiments/electric-replication/evidence'
origins=defaultdict(set)
for row in json.loads((OUT/'map.json').read_text()):
    origins[row['projection_sha256']].add(row['path'])
api_digest=hashlib.sha256((ROOT/'experiments/electric-replication/engine/src/api.rs').read_bytes()).hexdigest()
assert (OUT/'gitleaks-exit.txt').read_text().strip()=='1'
findings=json.loads((OUT/'gitleaks.json').read_text())
assert findings
verified_source=verified_object=0
reviewed=[]
for item in findings:
    assert item['RuleID']=='generic-api-key' and item['Tags']==[]
    assert item['Secret']=='REDACTED' and item['StartLine']==item['EndLine']
    file=ROOT/item['File']
    data=file.read_bytes()
    assert hashlib.sha256(data).hexdigest()==file.stem
    paths=origins[file.stem]
    assert paths
    line=data.decode().splitlines()[item['StartLine']-1]
    if item['Match']=='api.rs": "REDACTED"':
        match=re.fullmatch(r'\s*"engine/src/api.rs": "([0-9a-f]{64})",?',line)
        assert match and match[1]==api_digest
        verified_source+=1
        kind='recomputed-source-sha256'
    else:
        assert item['Match']=='key_sha256": "REDACTED"'
        path=EVIDENCE/'cold-work-harness-repair-001/diagnosis.json'
        assert paths=={str(path.relative_to(ROOT))}
        match=re.fullmatch(r'\s*"key_sha256": "([0-9a-f]{64})",?',line)
        assert match
        diagnosis=json.loads(path.read_text())['retained_failures']['candidate-quorum3-r2']['late_completed_deletes']
        assert len(diagnosis)==1 and diagnosis[0]['key_sha256']==match[1]
        cell=EVIDENCE/'cold-work-generations-001/cold-work-generations-001-candidate-quorum3-r2'
        result=json.loads((cell/'result.json').read_text())
        cycle=next(c for c in result['cycles'] if c['generation']==diagnosis[0]['generation'])
        keys={'/'+result['object_bucket']+'/'+key for key in cycle['precondition']['listing_keys']}
        matches=[key for key in keys if hashlib.sha256(key.encode()).hexdigest()==match[1]]
        assert len(matches)==1
        events=[e for e in read_object_events(cell) if e.get('op')=='s3' and e['path']==matches[0]
                and e['method']=='DELETE' and e['status']==204 and e['time_ns']>cycle['collected_ns']]
        assert len(events)==1 and events[0]['time_ns']-cycle['collected_ns']==diagnosis[0]['completion_minus_collection_ns']
        verified_object+=1
        kind='recomputed-public-object-name-sha256'
    reviewed.append(dict(kind=kind,paths=sorted(paths),line=item['StartLine']))
result=dict(verdict='PASS_REVIEWED_FINDINGS',scanner='Gitleaks 8.30.1 default rules; no exclusions',
    scanner_exit=1,findings=len(findings),independently_verified_source_sha256=verified_source,
    independently_verified_public_object_key_sha256=verified_object,unexplained=0,reviewed=reviewed,
    method='Every complete reported line must match its expected hash-only grammar. Source digests are recomputed from actual source bytes. The diagnosis object hash is recomputed from the independently listed public object name and checked against the real completed DELETE event. No finding is waived by its filename or an old expected finding count.',
    report_sha256=hashlib.sha256((OUT/'gitleaks.json').read_bytes()).hexdigest())
(OUT/'findings-review.json').write_text(json.dumps(result,indent=2)+'\n')
print(json.dumps({k:v for k,v in result.items() if k!='reviewed'},indent=2))
